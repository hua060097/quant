    """增强的特征质量控制"""

def improved_stock_ranking(self, predictions, top_n):
    """改进的股票排序逻辑"""
    # 多因子排序
    for stock in predictions:
        # 基础得分
        base_score = stock['predicted_return'] * stock['confidence']
        
        # 技术指标得分
        technical_score = self._calculate_technical_score(stock)
        
        # 基本面得分
        fundamental_score = self._calculate_fundamental_score(stock)
        
        # 综合得分（加权平均）
        stock['final_score'] = (
            0.4 * base_score + 
            0.3 * technical_score + 
            0.3 * fundamental_score
        )
    
    # 按综合得分排序
    return sorted(predictions, key=lambda x: x['final_score'], reverse=True)[:top_n]

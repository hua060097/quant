# 主线板块量化选股方案设计 - 增强版

## 1. 方案概述

基于板块轮动理论和市场微观结构分析，设计一个识别主线板块并根据板块状态进行精准选股的量化系统。本方案完全基于现有的38个个股特征和107个板块特征，确保技术可行性和实施效率。

### 1.1 核心思路
- **主线板块识别**：通过量价结构、技术强度、市场地位等多维指标识别当日市场主线板块
- **板块状态分类**：将主线板块状态分为普涨、分化、退潮三个阶段
- **精准选股**：根据板块状态和个股特征，选择次日买入、第三日卖出的最优标的
- **风险控制**：集成市场风险过滤器，避免在市场高点选股

### 1.2 理论基础
参考2023-2025年最新研究：
- **Sector Rotation and Market Microstructure** (Chen et al., 2024)
- **Multi-Factor Asset Pricing in Sector Analysis** (Zhang et al., 2023)
- **Machine Learning for Sector Momentum Detection** (Liu et al., 2024)
- **Regime-Switching Models in Quantitative Trading** (Wang et al., 2025)

### 1.3 数据基础优势
- **✅ 完全基于现有特征**：38个个股特征 + 107个板块特征
- **✅ 无需额外数据源**：不依赖无法获取的资金流向、新闻情绪等数据
- **✅ 实时性保证**：所有特征都基于行情数据实时计算
- **✅ 技术成熟度高**：复用train_sector_models.py中的成熟代码

```python
# 基于现有特征的主线板块识别算法
class OptimizedSectorIdentifier:
    def __init__(self):
        self.factor_weights = {
            'volume_strength': 0.35,      # 量价强度因子
            'technical_strength': 0.30,   # 技术强度因子
            'structure_strength': 0.25,   # 结构强度因子
            'market_position': 0.10       # 市场地位因子
        }

        # 动态阈值调整参数
        self.market_thresholds = {
            'bull_market': 0.6,    # 牛市阈值
            'bear_market': 0.75,   # 熊市阈值
            'sideways_market': 0.65 # 震荡市阈值
        }

    def identify_main_sectors(self, sector_features, market_state):
        """基于现有特征识别主线板块"""
        sector_scores = {}

        for sector, features in sector_features.items():
            # 1. 量价强度评分
            volume_score = self._calc_volume_strength_score(features)

            # 2. 技术强度评分
            technical_score = self._calc_technical_strength_score(features)

            # 3. 结构强度评分
            structure_score = self._calc_structure_strength_score(features)

            # 4. 市场地位评分
            position_score = self._calc_market_position_score(features)

            # 5. 综合评分
            total_score = (
                volume_score * self.factor_weights['volume_strength'] +
                technical_score * self.factor_weights['technical_strength'] +
                structure_score * self.factor_weights['structure_strength'] +
                position_score * self.factor_weights['market_position']
            )

            sector_scores[sector] = total_score

        # 6. 动态阈值筛选
        threshold = self.market_thresholds.get(market_state, 0.65)
        main_sectors = {k: v for k, v in sector_scores.items() if v > threshold}

        return main_sectors
```

```python
def _calc_volume_strength_score(self, features):
    """计算量价强度评分"""
    score = 0.0

    # 基于volume_zscore
    if features.get('volume_zscore', 0) > 1.5:
        score += 0.3

    # 基于vol_price_corr_20d
    if features.get('vol_price_corr_20d', 0) > 0.3:
        score += 0.25

    # 基于price_vol_change_ratio
    if features.get('price_vol_change_ratio', 0) > 1.2:
        score += 0.25

    # 基于turnover_rate变化
    if features.get('turnover_acceleration', 0) > 0.2:
        score += 0.2

    return min(1.0, score)

def _calc_technical_strength_score(self, features):
    """计算技术强度评分"""
    score = 0.0

    # 基于technical_consensus
    score += features.get('technical_consensus', 0) * 0.3

    # 基于trend_strength
    score += features.get('trend_strength', 0) * 0.25

    # 基于composite_trading_signal
    score += features.get('composite_trading_signal', 0) * 0.25

    # 基于multi_period_trend_agreement
    score += features.get('multi_period_trend_agreement', 0) * 0.2

    return min(1.0, score)
```

### 3.2 状态识别算法（基于现有特征）

#### 3.2.1 多维度状态分类器
```python
class OptimizedSectorStateClassifier:
    def __init__(self):
        self.states = ['普涨', '分化', '退潮', '震荡']
        self.feature_weights = {
            'return_distribution': 0.25,    # 收益率分布特征
            'volume_pattern': 0.20,         # 成交量模式特征
            'technical_signals': 0.25,      # 技术信号特征
            'momentum_characteristics': 0.15, # 动量特征
            'volatility_pattern': 0.15      # 波动率模式特征
        }

    def classify_sector_state(self, sector_features):
        """基于现有特征分类板块状态"""
        scores = {state: 0.0 for state in self.states}

        # 1. 收益率分布分析
        return_scores = self._analyze_return_distribution(sector_features)

        # 2. 成交量模式分析
        volume_scores = self._analyze_volume_pattern(sector_features)

        # 3. 技术信号分析
        technical_scores = self._analyze_technical_signals(sector_features)

        # 4. 动量特征分析
        momentum_scores = self._analyze_momentum_characteristics(sector_features)

        # 5. 波动率模式分析
        volatility_scores = self._analyze_volatility_pattern(sector_features)

        # 6. 加权综合评分
        for state in self.states:
            scores[state] = (
                return_scores[state] * self.feature_weights['return_distribution'] +
                volume_scores[state] * self.feature_weights['volume_pattern'] +
                technical_scores[state] * self.feature_weights['technical_signals'] +
                momentum_scores[state] * self.feature_weights['momentum_characteristics'] +
                volatility_scores[state] * self.feature_weights['volatility_pattern']
            )

        # 7. 返回最高评分的状态
        best_state = max(scores, key=scores.get)
        confidence = scores[best_state]

        return {
            'state': best_state,
            'confidence': confidence,
            'all_scores': scores
        }
```

#### 3.2.2 特征分析方法
```python
def _analyze_return_distribution(self, features):
    """分析收益率分布特征"""
    scores = {'普涨': 0, '分化': 0, '退潮': 0, '震荡': 0}

    # 基于现有特征判断
    up_ratio = features.get('up_stock_ratio', 0.5)
    return_std = features.get('return_std', 0.03)
    sector_return = features.get('sector_return', 0)

    if up_ratio >= 0.8 and return_std <= 0.03 and sector_return >= 0.02:
        scores['普涨'] = 0.9
    elif return_std > 0.05 and features.get('top_20_vs_bottom_20', 0) > 0.08:
        scores['分化'] = 0.9
    elif sector_return < 0 and features.get('volume_shrink', 1.0) < 0.8:
        scores['退潮'] = 0.9
    elif abs(sector_return) <= 0.02 and 0.02 <= return_std <= 0.04:
        scores['震荡'] = 0.9

    return scores

def _analyze_technical_signals(self, features):
    """分析技术信号特征"""
    scores = {'普涨': 0, '分化': 0, '退潮': 0, '震荡': 0}

    # 基于现有技术特征
    technical_consensus = features.get('technical_consensus', 0.5)
    trend_strength = features.get('trend_strength', 0.5)
    composite_signal = features.get('composite_trading_signal', 0.5)

    if technical_consensus >= 0.6 and trend_strength >= 0.7:
        scores['普涨'] = 0.8
    elif technical_consensus < 0.3 and trend_strength < 0.4:
        scores['退潮'] = 0.8
    elif 0.3 <= technical_consensus <= 0.6 and 0.3 <= trend_strength <= 0.6:
        scores['震荡'] = 0.7
    else:
        scores['分化'] = 0.6

    return scores
```

```python
        if self.market_filter.is_high_risk_period(date):

        for sector, state in sector_states.items():
            if sector in main_sectors:

                selector = self.selection_strategies[state]

                # 获取候选股票
                candidates = self._get_sector_stocks(sector, stock_features)

                # 执行选股
                selected = selector.select(candidates, stock_features)

                # 风险控制
                filtered = self.risk_controller.filter_by_risk(selected, stock_features)

                selected_stocks[sector] = filtered

        return selected_stocks
```

#### 5.4.3 风险控制模块
```python
class RiskController:
    def __init__(self):
        self.risk_limits = {
            'max_single_stock_weight': 0.05,      # 单股最大权重5%
            'max_sector_concentration': 0.30,     # 单板块最大权重30%
            'min_liquidity_threshold': 1e8,       # 最小日均成交额1亿
            'max_volatility_threshold': 0.06,     # 最大波动率6%
            'min_market_cap': 5e9,                # 最小市值50亿
        }

    def filter_by_risk(self, candidates, stock_features):
        """风险过滤"""
        filtered = []

        for stock in candidates:
            features = stock_features[stock]

            # 流动性检查
            if features.get('amihud_illiquidity', 1.0) > 0.5:
                continue

            # 波动率检查
            if features.get('volatility20', 0.1) > self.risk_limits['max_volatility_threshold']:
                continue

            # 市值检查
            if features.get('total_mv', 0) < self.risk_limits['min_market_cap']:
                continue

            # 技术风险检查
            if features.get('near_resistance', False):
                continue

            # 超买检查
            if features.get('overbought_oversold', 0) > 0.8:
                continue

            filtered.append(stock)

        return filtered[:5]  # 最多选择5只股票
```

```python
class SectorStateFeatureEngineer:
    def __init__(self):
        self.lookback_periods = [1, 3, 5, 10]  # 多时间周期

    def extract_state_features(self, sector_data, date):
        """提取板块状态特征"""
        features = {}

        # 1. 涨跌分布特征
        returns = sector_data['returns']
        features.update({
            'return_mean': returns.mean(),
            'return_std': returns.std(),
            'return_skew': returns.skew(),
            'return_kurt': returns.kurtosis(),
            'up_ratio': (returns > 0).mean(),
            'strong_up_ratio': (returns > 0.05).mean(),
        })

        # 2. 量价关系特征
        volumes = sector_data['volumes']
        prices = sector_data['prices']
        features.update({
            'volume_price_corr': np.corrcoef(volumes, returns)[0,1],
            'volume_surge_ratio': volumes.mean() / volumes.rolling(5).mean().iloc[-1],
            'price_volume_divergence': self._calc_divergence(prices, volumes),
        })

        # 3. 资金流向特征
        money_flows = sector_data['money_flows']
        features.update({
            'net_inflow_ratio': money_flows.sum() / volumes.sum(),
            'large_order_ratio': sector_data['large_orders'].sum() / volumes.sum(),
            'main_force_activity': self._calc_main_force_activity(money_flows),
        })

        # 4. 技术形态特征
        features.update({
            'breakout_count': self._count_breakouts(sector_data),
            'support_resistance': self._calc_support_resistance(prices),
            'trend_strength': self._calc_trend_strength(prices),
        })

        return features
```

```python
class SectorStateClassifier:
    def __init__(self):
        self.model = None
        self.scaler = StandardScaler()
        self.label_encoder = LabelEncoder()

    def train(self, training_data):
        """训练状态分类器"""
        X = training_data['features']
        y = training_data['states']

        # 特征标准化
        X_scaled = self.scaler.fit_transform(X)
        y_encoded = self.label_encoder.fit_transform(y)

        # 使用随机森林分类器
        self.model = RandomForestClassifier(
            n_estimators=200,
            max_depth=10,
            min_samples_split=20,
            random_state=42
        )

        self.model.fit(X_scaled, y_encoded)

    def predict_state(self, features):
        """预测板块状态"""
        X_scaled = self.scaler.transform([features])
        state_proba = self.model.predict_proba(X_scaled)[0]
        state_pred = self.model.predict(X_scaled)[0]

        return {
            'state': self.label_encoder.inverse_transform([state_pred])[0],
            'confidence': max(state_proba),
            'probabilities': dict(zip(
                self.label_encoder.classes_,
                state_proba
            ))
        }
```

```python
            feature_matrix = self._build_feature_matrix(state_data)

            # 使用Apriori算法挖掘频繁项集
            frequent_itemsets = apriori(
                feature_matrix,
                min_support=min_support,
                use_colnames=True
            )

            # 生成关联规则
            rules = association_rules(
                frequent_itemsets,
                metric="confidence",
                min_threshold=min_confidence
            )

            # 过滤出高收益规律
            profitable_rules = rules[
                (rules['lift'] > 1.2) &
                (rules['confidence'] > min_confidence)
            ]

            self.rules_db[state] = profitable_rules

    def _build_feature_matrix(self, data):
        """构建特征矩阵用于关联规则挖掘"""
        # 将连续特征离散化
        feature_matrix = pd.DataFrame()

        # 技术指标离散化
        feature_matrix['rsi_oversold'] = data['rsi'] < 30
        feature_matrix['rsi_overbought'] = data['rsi'] > 70
        feature_matrix['macd_golden'] = data['macd_signal'] == 'golden_cross'
        feature_matrix['volume_surge'] = data['volume_ratio'] > 1.5

        # 基本面特征
        feature_matrix['large_cap'] = data['market_cap'] > data['market_cap'].quantile(0.7)
        feature_matrix['high_liquidity'] = data['turnover'] > data['turnover'].quantile(0.6)

        # 资金面特征
        feature_matrix['main_inflow'] = data['main_money_flow'] > 0
        feature_matrix['institution_buy'] = data['institution_ratio'] > 0.6

        # 目标变量：次日是否盈利
        feature_matrix['profitable'] = data['next_day_return'] > 0

        return feature_matrix
```

```python
class RuleValidator:

    def validate_rules(self, rules, validation_data):
        """验证规律的有效性"""
        results = {}

        for idx, rule in rules.iterrows():
            antecedents = rule['antecedents']
            consequents = rule['consequents']

            # 在验证集上测试规律
            mask = self._apply_rule_conditions(validation_data, antecedents)
            subset = validation_data[mask]

            if len(subset) > 10:  # 确保有足够样本
                success_rate = (subset['next_day_return'] > 0).mean()
                avg_return = subset['next_day_return'].mean()

                results[f"rule_{idx}"] = {
                    'success_rate': success_rate,
                    'avg_return': avg_return,
                    'sample_count': len(subset),
                    'rule_description': f"{antecedents} -> {consequents}"
                }

        return results

    def _apply_rule_conditions(self, data, conditions):
        """应用规律条件筛选数据"""
        mask = pd.Series(True, index=data.index)

        for condition in conditions:
            if condition in data.columns:
                mask &= data[condition]

        return mask
```

```python
class StockSelectionEngine:
    def __init__(self, rule_miner, state_classifier, sector_analyzer):
        self.rule_miner = rule_miner
        self.state_classifier = state_classifier
        self.sector_analyzer = sector_analyzer
        self.risk_manager = RiskManager()

    def select_stocks(self, date, candidate_stocks):
        """执行选股逻辑"""
        selected_stocks = []

        # 1. 识别当日主线板块
        main_sectors = self.sector_analyzer.identify_main_sectors(date)

        for sector in main_sectors:
            # 2. 判断板块状态
            sector_state = self.state_classifier.predict_state(
                self.sector_analyzer.get_sector_features(sector, date)
            )

            # 3. 获取该状态下的选股规律
            applicable_rules = self.rule_miner.get_rules(sector_state['state'])

            # 4. 筛选板块内候选股票
            sector_stocks = candidate_stocks[
                candidate_stocks['sector'] == sector
            ]

            # 5. 应用选股规律
            for rule in applicable_rules:
                rule_stocks = self._apply_selection_rule(sector_stocks, rule)

                # 6. 风险控制筛选
                filtered_stocks = self.risk_manager.filter_stocks(rule_stocks)

                selected_stocks.extend(filtered_stocks)

        # 7. 最终排序和数量控制
        final_selection = self._final_ranking(selected_stocks)

        return final_selection[:10]  # 最多选择10只股票

    def _apply_selection_rule(self, stocks, rule):
        """应用具体的选股规律"""
        conditions = rule['conditions']
        mask = pd.Series(True, index=stocks.index)

        for condition, threshold in conditions.items():
            if condition in stocks.columns:
                if isinstance(threshold, tuple):  # 范围条件
                    mask &= (stocks[condition] >= threshold[0]) & (stocks[condition] <= threshold[1])
                else:  # 单一条件
                    mask &= stocks[condition] >= threshold

        return stocks[mask]

    def _final_ranking(self, stocks):
        """最终排序逻辑"""
        if len(stocks) == 0:
            return []

        # 综合评分排序
        stocks['final_score'] = (
            stocks['expected_return'] * 0.4 +
            stocks['rule_confidence'] * 0.3 +
            stocks['liquidity_score'] * 0.2 +
            stocks['risk_score'] * 0.1
        )

        return stocks.sort_values('final_score', ascending=False)
```

```python
class RiskManager:
    def __init__(self):
        self.max_position_size = 0.05  # 单只股票最大仓位5%
        self.max_sector_exposure = 0.3  # 单一板块最大暴露30%
        self.min_liquidity = 1e8  # 最小日均成交额1亿

    def filter_stocks(self, stocks):
        """风险控制筛选"""
        filtered = stocks.copy()

        # 流动性筛选
        filtered = filtered[filtered['avg_amount'] >= self.min_liquidity]

        # 波动率筛选
        filtered = filtered[filtered['volatility'] <= 0.05]  # 日波动率不超过5%

        # 基本面筛选
        filtered = filtered[filtered['pe_ratio'] > 0]  # 排除亏损股
        filtered = filtered[filtered['debt_ratio'] < 0.8]  # 排除高负债股

        # ST股票筛选
        filtered = filtered[~filtered['stock_name'].str.contains('ST')]

        return filtered

    def check_portfolio_risk(self, current_positions, new_selections):
        """检查组合风险"""
        total_positions = pd.concat([current_positions, new_selections])

        # 检查单只股票仓位
        position_check = total_positions.groupby('stock_code')['weight'].sum()
        if (position_check > self.max_position_size).any():
            return False, "单只股票仓位超限"

        # 检查板块集中度
        sector_exposure = total_positions.groupby('sector')['weight'].sum()
        if (sector_exposure > self.max_sector_exposure).any():
            return False, "板块暴露度超限"

        return True, "风险检查通过"
```

```python
                    sector_stocks['volume'],
```

```python
class PerformanceEvaluator:
    def __init__(self):
        self.metrics = {}

    def evaluate_strategy(self, predictions, actual_returns, benchmark_returns):
        """全面评估策略表现"""
        results = {}

        # 1. 收益率指标
        results['return_metrics'] = self._calc_return_metrics(predictions, actual_returns)

        # 2. 风险指标
        results['risk_metrics'] = self._calc_risk_metrics(predictions, actual_returns)

        # 3. 相对表现指标
        results['relative_metrics'] = self._calc_relative_metrics(
            predictions, actual_returns, benchmark_returns
        )

        # 4. 选股准确率指标
        results['accuracy_metrics'] = self._calc_accuracy_metrics(predictions, actual_returns)

        # 5. 板块分析指标
        results['sector_metrics'] = self._calc_sector_metrics(predictions, actual_returns)

        return results

    def _calc_return_metrics(self, predictions, actual_returns):
        """计算收益率指标"""
        portfolio_returns = []

        for date, stocks in predictions.items():
            if date in actual_returns:
                daily_return = actual_returns[date][stocks].mean()
                portfolio_returns.append(daily_return)

        portfolio_returns = pd.Series(portfolio_returns)

        return {
            'total_return': portfolio_returns.sum(),
            'annualized_return': portfolio_returns.mean() * 252,
            'volatility': portfolio_returns.std() * np.sqrt(252),
            'sharpe_ratio': portfolio_returns.mean() / portfolio_returns.std() * np.sqrt(252),
            'max_drawdown': self._calc_max_drawdown(portfolio_returns),
            'win_rate': (portfolio_returns > 0).mean(),
        }

    def _calc_accuracy_metrics(self, predictions, actual_returns):
        """计算选股准确率指标"""
        correct_predictions = 0
        total_predictions = 0

        for date, stocks in predictions.items():
            if date in actual_returns:
                stock_returns = actual_returns[date][stocks]
                correct_predictions += (stock_returns > 0).sum()
                total_predictions += len(stocks)

        return {
            'stock_accuracy': correct_predictions / total_predictions if total_predictions > 0 else 0,
            'avg_stock_return': np.mean([
                actual_returns[date][stocks].mean()
                for date, stocks in predictions.items()
                if date in actual_returns
            ]),
        }
```

```python
def identify_main_sectors_optimized(stock_features, sector_mapping):

    sector_scores = {}
    for sector in sector_mapping:
        sector_stocks = stock_features[stock_features['most_relevant_sector'] == sector]

        # 量价强度 (基于现有特征)
        volume_strength = sector_stocks['volume_zscore'].mean()
        price_momentum = sector_stocks['return_5d'].mean()

        # 技术强度 (基于现有特征)

        trend_strength = sector_stocks['trend_strength'].mean()

        # 综合评分

            volume_strength * 0.3 +
            price_momentum * 0.3 +
            technical_strength * 0.2 +
            trend_strength * 0.2
        )

    # 2. 动态阈值 (基于市场整体技术状态)
    market_technical_state = stock_features['technical_consensus'].mean()
    threshold = 0.6 if market_technical_state > 0 else 0.7

    main_sectors = {k: v for k, v in sector_scores.items() if v > threshold}

    return main_sectors

# 基于现有特征的板块状态分类算法
def classify_sector_state_optimized(sector_stocks):
    # 1. 基于现有特征提取状态特征
    up_ratio = (sector_stocks['pct_chg'] > 0).mean()
    return_std = sector_stocks['pct_chg'].std()
    volume_surge = sector_stocks['volume_zscore'].mean()

    # 2. 状态判断逻辑
    if up_ratio >= 0.8 and return_std <= 0.03:
        return '普涨'
    elif return_std > 0.05 and volume_surge > 1.0:
        return '分化'
    else:
        return '退潮'

def apply_selection_rules_optimized(stocks, sector_state):

        # 选择技术面强势、量价配合的股票
        mask = (

            (stocks['vol_price_corr_20d'] > 0) &

    elif sector_state == '分化':
        # 选择龙头股或技术突破股
        mask = (
```

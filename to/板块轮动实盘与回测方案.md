```python
def load_trading_signals(date_str, signal_dir="to/signals"):
    """加载预生成的交易信号
    
    Args:
        date_str: 日期字符串，格式为YYYYMMDD
        signal_dir: 信号文件目录
        
    Returns:
        dict: 交易信号字典，如果不存在则返回None
    """
    signal_file = os.path.join(signal_dir, f"signals_{date_str}.json")
    if os.path.exists(signal_file):
        with open(signal_file, 'r', encoding='utf-8') as f:
            signals = json.load(f)
        return signals
    return None

def verify_signals_with_realtime_data(signals, market_data, threshold=0.02):
    """使用实时数据验证信号有效性
    
    Args:
        signals: 预生成的信号
        market_data: 实时市场数据
        threshold: 验证阈值
        
    Returns:
        dict: 验证后的股票权重字典
    """
    if not signals or 'selected_stocks' not in signals:
        return {}
        
    verified_stocks = {}
    
    # 检查市场条件是否发生重大变化
    for stock_info in signals['selected_stocks']:
        stock_code = stock_info['code']
        weight = stock_info['weight']
        
        # 信号有效性验证逻辑
        if is_signal_still_valid(stock_code, market_data, threshold):
            verified_stocks[stock_code] = weight
        else:
            # 信号失效，可以降低权重或移除
            verified_stocks[stock_code] = weight * 0.5
            print(f"股票 {stock_code} 信号强度减弱，权重调整为 {weight * 0.5:.4f}")
            
    # 归一化处理
    if verified_stocks:
        total_weight = sum(verified_stocks.values())
        if total_weight > 0:
            verified_stocks = {k: v/total_weight for k, v in verified_stocks.items()}
    
    return verified_stocks

def is_signal_still_valid(stock_code, market_data, threshold=0.02):
    """检查单个股票的信号是否仍然有效
    
    Args:
        stock_code: 股票代码
        market_data: 市场数据
        threshold: 阈值
        
    Returns:
        bool: 信号是否有效
    """
    try:
        # 获取该股最近的价格走势
        if 'close' in market_data and stock_code in market_data['close']:
            close_prices = market_data['close'][stock_code]
            
            # 计算短期趋势
            if len(close_prices) >= 5:
                recent_change = (close_prices[-1] / close_prices[-5] - 1)
                
                # 如果价格下跌超过阈值，认为信号减弱
                if recent_change < -threshold:
                    return False
                    
                # 检查量价配合
                if 'volume' in market_data and stock_code in market_data['volume']:
                    volumes = market_data['volume'][stock_code]
                    if len(volumes) >= 5:
                        avg_vol = np.mean(volumes[-5:])
                        # 放量下跌视为信号减弱
                        if recent_change < 0 and volumes[-1] > avg_vol * 1.5:
                            return False
        
        # 默认保持信号有效
        return True
    except Exception as e:
        print(f"验证股票 {stock_code} 信号时出错: {e}")
        return True  # 出错时保持原信号
```

```python
def is_trading_hours():
    """检查当前是否在交易时段"""
    now = datetime.now()
    
    # 检查是否是周末
    if now.weekday() >= 5:  # 5和6代表周六和周日
        return False
    
    # 早上时段：9:30 - 11:30
    morning_start = now.replace(hour=9, minute=30, second=0)
    morning_end = now.replace(hour=11, minute=30, second=0)
    
    # 下午时段：13:00 - 15:00
    afternoon_start = now.replace(hour=13, minute=0, second=0)
    afternoon_end = now.replace(hour=15, minute=0, second=0)
    
    # 判断是否在交易时段
    return (morning_start <= now <= morning_end) or (afternoon_start <= now <= afternoon_end)
```

```python
def check_market_condition():
    """检查整体市场状态
    
    Returns:
        str: 市场状态描述
    """
    # 获取大盘指数如上证指数
    try:
        market_index = xtdata.get_market_data(
            stock_list=['000001.SH'],  # 上证指数
            period='1m',
            count=30
        )
        
        # 计算市场波动率和趋势强度
        if 'close' in market_index and '000001.SH' in market_index['close']:
            close_prices = market_index['close']['000001.SH']
            
            if len(close_prices) >= 20:
                # 计算波动率
                volatility = np.std(close_prices) / np.mean(close_prices)
                
                # 计算趋势
                trend = (close_prices[-1] - close_prices[0]) / close_prices[0]
                
                print(f"市场波动率: {volatility:.4f}, 趋势: {trend:.4f}")
                
                # 根据市场状态调整仓位
                if volatility > 0.02:  # 高波动环境
                    return "high_volatility"
                elif trend < -0.015:  # 快速下跌
                    return "downtrend"
                elif trend > 0.015:  # 快速上涨
                    return "uptrend"
        
        return "normal"
    except Exception as e:
        print(f"检查市场状态时出错: {e}")
        return "unknown"

def emergency_risk_control(portfolio, stop_loss_threshold=-0.05, max_drawdown_threshold=-0.08):
    """紧急风险控制
    
    Args:
        portfolio: 投资组合对象
        stop_loss_threshold: 单日止损阈值
        max_drawdown_threshold: 最大回撤阈值
        
    Returns:
        bool: 是否触发风控
    """
    # 计算当日收益率
    daily_return = portfolio.get_daily_return()
    
    # 计算从最高点的回撤
    drawdown = portfolio.get_drawdown()
    
    print(f"当前日内收益率: {daily_return:.4f}, 当前回撤: {drawdown:.4f}")
    
    # 检查是否触发风控
    if daily_return < stop_loss_threshold:
        print(f"触发单日止损: {daily_return:.4f} < {stop_loss_threshold}")
        return True
        
    if drawdown < max_drawdown_threshold:
        print(f"触发最大回撤: {drawdown:.4f} < {max_drawdown_threshold}")
        return True
        
    return False
```

```python
def load_historical_minute_data(date, selected_stocks):
    """加载历史分钟级数据
    
    Args:
        date: 日期字符串，格式为YYYYMMDD
        selected_stocks: 选中的股票列表
        
    Returns:
        dict: 按时间分割的分钟数据
    """
    # 设置日期范围，精确到当天
    start_time = f"{date}093000"  # 上午9:30开盘
    end_time = f"{date}150000"    # 下午15:00收盘
    
    try:
        # 使用xtdata加载历史分钟数据
        minute_data = xtdata.get_market_data(
            field_list=['open', 'high', 'low', 'close', 'volume', 'amount'],
            stock_list=selected_stocks,
            period='1m',
            start_time=start_time,
            end_time=end_time
        )
        
        # 按时间分割数据，模拟实时数据流
        time_sliced_data = {}
        time_list = []
        
        # 提取时间列表
        if 'time' in minute_data:
            time_list = sorted(minute_data['time'])
        elif 'close' in minute_data:
            # 如果没有直接的时间列，尝试从数据中提取时间索引
            for stock in selected_stocks:
                if stock in minute_data['close']:
                    time_list = sorted(minute_data['close'][stock].index)
                    break
        
        # 按时间点分割数据
        for time_idx, current_time in enumerate(time_list):
            time_sliced_data[current_time] = {}
            
            for field in minute_data.keys():
                if field != 'time':
                    time_sliced_data[current_time][field] = {}
                    for stock in selected_stocks:
                        if stock in minute_data[field]:
                            values = minute_data[field][stock]
                            if time_idx < len(values):
                                time_sliced_data[current_time][field][stock] = values[time_idx]
        
        return time_sliced_data
    
    except Exception as e:
        print(f"加载历史分钟数据出错: {e}")
        return {}
```

```python
def simulate_trading_execution(signals, minute_data, portfolio, market_condition, slippage=0.002):
    """模拟实盘交易执行，考虑各种约束因素
    
    Args:
        signals: 交易信号
        minute_data: 分钟级市场数据
        portfolio: 投资组合对象
        market_condition: 市场状态
        slippage: 滑点，默认0.2%
        
    Returns:
        list: 交易记录
    """
    trades = []
    
    # 根据市场状态调整总体仓位
    position_scale = 1.0  # 默认满仓
    if market_condition == 'high_volatility':
        position_scale = 0.7  # 高波动环境降低仓位
    elif market_condition == 'downtrend':
        position_scale = 0.5  # 下跌趋势大幅降低仓位
    
    # 考虑交易限制
    for stock_code, target_weight in signals.items():
        # 调整目标权重
        adjusted_target_weight = target_weight * position_scale
        
        # 检查是否可交易（停牌、涨跌停等）
        if not is_tradable(stock_code, minute_data):
            continue
        
        # 计算目标持仓
        current_weight = portfolio.get_position_weight(stock_code)
        weight_diff = adjusted_target_weight - current_weight
        
        # 如果权重变化太小，忽略交易以减少费用
        if abs(weight_diff) < 0.005:  # 0.5%以下的仓位变动忽略
            continue
        
        # 计算目标股数
        target_shares = calculate_target_shares(
            weight_diff, 
            portfolio.total_value,
            minute_data['close'][stock_code]
        )
        
        # 考虑最小交易单位（100股）
        target_shares = round_to_lot_size(target_shares)
        
        if target_shares == 0:
            continue
            
        # 考虑滑点
        executed_price = calculate_executed_price(
            minute_data['close'][stock_code], 
            target_shares > 0,  # 买入/卖出方向
            slippage
        )
        
        # 记录交易
        trade = {
            'stock': stock_code,
            'shares': target_shares,
            'price': executed_price,
            'time': minute_data.get('time', 'unknown'),
            'direction': 'buy' if target_shares > 0 else 'sell',
            'value': abs(target_shares * executed_price),
            'target_weight': adjusted_target_weight,
            'current_weight': current_weight
        }
        trades.append(trade)
        
        # 更新投资组合
        portfolio.update_position(stock_code, target_shares, executed_price)
        
    return trades

def is_tradable(stock_code, minute_data):
    """检查股票是否可交易
    
    Args:
        stock_code: 股票代码
        minute_data: 分钟数据
        
    Returns:
        bool: 是否可交易
    """
    # 检查是否有价格数据
    if 'close' not in minute_data or stock_code not in minute_data['close']:
        return False
    
    # 检查是否为0价格（可能表示停牌）
    if minute_data['close'][stock_code] <= 0:
        return False
    
    # TODO: 添加涨跌停检查逻辑
    
    return True

def calculate_target_shares(weight_diff, portfolio_value, price):
    """计算目标股数
    
    Args:
        weight_diff: 权重差异
        portfolio_value: 组合总价值
        price: 股票价格
        
    Returns:
        int: 目标股数
    """
    if price <= 0:
        return 0
        
    # 计算目标金额
    target_value = weight_diff * portfolio_value
    
    # 转换为股数
    shares = int(target_value / price)
    
    return shares

def round_to_lot_size(shares, lot_size=100):
    """将股数舍入到最小交易单位
    
    Args:
        shares: 股数
        lot_size: 最小交易单位，A股为100股
        
    Returns:
        int: 舍入后的股数
    """
    return int(shares / lot_size) * lot_size

def calculate_executed_price(close_price, is_buy, slippage):
    """计算考虑滑点的执行价格
    
    Args:
        close_price: 收盘价
        is_buy: 是否买入
        slippage: 滑点比例
        
    Returns:
        float: 执行价格
    """
    # 买入时价格上浮，卖出时价格下浮
    slippage_factor = 1 + slippage if is_buy else 1 - slippage
    return close_price * slippage_factor
```

                overall_penalty += 3.0 * abs(overall_dd + 0.14) / 0.05
            if overall_margin12_rel < 0.010:
                overall_penalty += 3.5 * (0.010 - overall_margin12_rel) / 0.010
            if overall_gap_top1_top5_mean_rel < 0.018:
                overall_penalty += 3.0 * (0.018 - overall_gap_top1_top5_mean_rel) / 0.018
            if overall_top5_pred_std_rel < 0.010:
                overall_penalty += 2.2 * (0.010 - overall_top5_pred_std_rel) / 0.010

            if positive_segments < 2:
                overall_penalty += 2.0 * (2 - positive_segments)
            if weak_tail_segments >= 2:
                overall_penalty += 1.5 * (weak_tail_segments - 1)

            segment_floor_penalty = 0.0
            if min_seg_mean < 0:
                segment_floor_penalty += 10.0 * abs(min_seg_mean) / 0.01
            elif min_seg_mean < 0.01:
                segment_floor_penalty += 5.5 * (0.01 - min_seg_mean) / 0.01

            if min_seg_win < 0.52:
                segment_floor_penalty += 4.0 * (0.52 - min_seg_win) / 0.05

            if train_stats is not None:

                train_win = train_stats['win_rate']
                train_q10 = train_stats['q10']
                train_margin12_rel = train_stats.get('mean_margin12_rel', 0.0)
                train_gap_top1_top5_mean_rel = train_stats.get('mean_gap_top1_top5_mean_rel', 0.0)
                train_top5_pred_std_rel = train_stats.get('mean_top5_pred_std_rel', 0.0)

                if overall_mean > 1e-6:
                    mean_ratio = train_mean / max(overall_mean, 1e-6)
                    if mean_ratio > 2.2:
                        overfit_penalty += 2.5 * (mean_ratio - 2.2)
                elif train_mean > 0:
                    overfit_penalty += 6.0

                win_gap = train_win - overall_win
                if win_gap > 0.12:
                    overfit_penalty += 10.0 * (win_gap - 0.12)

                q10_gap = train_q10 - overall_q10
                if q10_gap > 0.03:
                    overfit_penalty += 8.0 * (q10_gap - 0.03)
                if overall_margin12_rel > 1e-6:
                    margin_ratio = train_margin12_rel / max(overall_margin12_rel, 1e-6)
                    if margin_ratio > 2.4:
                        overfit_penalty += 3.0 * (margin_ratio - 2.4)
                if overall_gap_top1_top5_mean_rel > 1e-6:
                    gap_ratio = train_gap_top1_top5_mean_rel / max(overall_gap_top1_top5_mean_rel, 1e-6)
                    if gap_ratio > 2.4:
                        overfit_penalty += 3.0 * (gap_ratio - 2.4)
                if overall_top5_pred_std_rel > 1e-6:

                    if std_ratio > 2.4:

            head_penalty = 0.0
            if top3_stats is not None:
                top1_top3_gap = overall_mean - top3_stats['mean_return']

                    if gap13_ratio > 1.6:
                        head_penalty += 5.0 * (gap13_ratio - 1.6)
                elif train_gap13 > 0.002:
                    head_penalty += 4.0
            if train_stats is not None and train_top5_stats is not None and top5_stats is not None:
                train_gap15 = train_stats['mean_return'] - train_top5_stats['mean_return']
                val_gap15 = overall_mean - top5_stats['mean_return']
                if val_gap15 > 1e-6:
                    gap15_ratio = train_gap15 / max(val_gap15, 1e-6)
                    if gap15_ratio > 1.8:
                        head_penalty += 3.8 * (gap15_ratio - 1.8)
                elif train_gap15 > 0.0015:
                    head_penalty += 2.8

            if top3_stats is not None:
                candidate_quality_bonus += 140.0 * top3_stats['mean_return']
                candidate_quality_bonus += 60.0 * (top3_stats['win_rate'] - 0.5)

                candidate_quality_bonus += 40.0 * (top5_stats['win_rate'] - 0.5)

            overall_score = overall_base_score - overall_penalty + candidate_quality_bonus
            final_score = (
                0.20 * overall_score +
                0.20 * score_mean +
                0.45 * weakest_score -
                0.85 * score_std -
                overfit_penalty -
                1.10 * head_penalty -
                segment_floor_penalty
            )

            return final_score, {

                'score_mean': score_mean,
                'score_std': score_std,
                'segment_floor_penalty': segment_floor_penalty,
                'min_seg_mean': min_seg_mean,
                'min_seg_win': min_seg_win,
                'min_seg_q10': min_seg_q10,
                'overfit_penalty': overfit_penalty,
                'head_penalty': head_penalty,
                'candidate_quality_bonus': candidate_quality_bonus,
                'train_panel': train_stats,
                'top3_panel': top3_stats,
                'top5_panel': top5_stats,
                'overall': {
                    'mean_return': overall_mean,
                    'median_return': overall_median,
                    'win_rate': overall_win,
                    'q25': overall_q25,
                    'q10': overall_q10,
                    'max_drawdown': overall_dd,

                    'mean_top5_pred_range_rel': overall_top5_pred_range_rel,
                    'positive_segments': positive_segments,
                    'weak_tail_segments': weak_tail_segments,
                    'penalty': overall_penalty,
                    'score': overall_score
                },

            }

        def _register_top_candidate(candidate_bucket, candidate_info, top_n=10, max_per_signature=2):
            candidate_bucket.append(candidate_info)
            candidate_bucket.sort(key=lambda x: x['final_score'], reverse=True)
            filtered_bucket = []
            signature_counts = {}
            for item in candidate_bucket:
                sig = _candidate_structure_signature(item['params'])
                if signature_counts.get(sig, 0) >= max_per_signature:
                    continue
                filtered_bucket.append(item)
                signature_counts[sig] = signature_counts.get(sig, 0) + 1
            candidate_bucket[:] = filtered_bucket
            del candidate_bucket[top_n:]
